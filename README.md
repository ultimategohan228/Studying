-- Содержательные термины:
-- 1. Участники -> Игроки (player): player_id, username
-- 2. Объекты -> Предметы (item): item_id, item_name
-- 3. Действия -> Покупки (player_item): player_item_id, player_id, item_id, quantity
--
-- Назначение объектов X и Y:
-- Выбранный объект X: Предмет 'Sword' (item_id = 3) -> Множество игроков А
-- Выбранный объект Y: Предмет 'Health Potion' (item_id = 6) -> Множество игроков B
-- ============================================================================

-- Запрос 1. Выберите конкретный тип объекта и числовую границу стоимости (в нашем случае - количества). 
-- Получите все атрибуты объектов этого типа со стоимостью (количеством) не выше указанной границы.
-- РА: σ[item_id = 1 AND quantity <= 5000](player_item)
-- Прогноз: Строки с player_item_id = 1 (quantity=1500) и player_item_id = 5 (quantity=50). Строка 33 (7777) отсеется.
SELECT player_item_id, player_id, item_id, quantity
FROM player_item
WHERE item_id = 1 AND quantity <= 5000;


-- Запрос 2. Получите перечень типов объектов без повторений (в нашей схеме - уникальные имена предметов).
-- РА: π[item_name](item)
-- Прогноз: 11 строк с уникальными названиями предметов (Gold, Diamond, Sword и т.д.)
SELECT DISTINCT item_name
FROM item;


-- Запрос 3. Получите журнал действий: ид_действия, наименование участника, название объекта, дата_действия (количество).
-- РА: π[player_item_id, username, item_name, quantity](player_item ⋈ player ⋈ item)
-- Прогноз: 35 строк, содержащих читаемые никнеймы и названия предметов вместо числовых внешних ключей.
SELECT pi.player_item_id, p.username, i.item_name, pi.quantity
FROM player_item pi
JOIN player p ON pi.player_id = p.player_id
JOIN item i ON pi.item_id = i.item_id;


-- Запрос 4. Выберите категорию участников (игроки с количеством предметов > 10) и тип объектов (Мечи, item_id=3). 
-- Найдите участников этой категории, совершавших действия с объектами указанного типа. Выведите идентификатор и наименование без повторений.
-- РА: π[player_id, username](σ[quantity > 10 AND item_id = 3](player ⋈ player_item))
-- Прогноз: 0 строк (В нашей базе игроки с мечами (id=3) имеют количество ровно 1, условие quantity > 10 отсечет их).
SELECT DISTINCT p.player_id, p.username
FROM player p
JOIN player_item pi ON p.player_id = pi.player_id
WHERE pi.quantity > 10 AND pi.item_id = 3;


-- Запрос 5. Объединение. Найдите участников, которые совершали действия с X или Y, включая тех, кто связан с обоими объектами. Используйте A ∪ B и UNION.
-- Множество А (брали Меч): player_id = 1, 7, 9
-- Множество B (брали Зелье здоровья): player_id = 1, 2, 5, 7, 9, 10
-- РА: π[player_id](σ[item_id=3](player_item)) ∪ π[player_id](σ[item_id=6](player_item))
-- Прогноз: Уникальный список ID игроков, купивших Меч или Зелье: 1, 2, 5, 7, 9, 10.
SELECT player_id FROM player_item WHERE item_id = 3
UNION
SELECT player_id FROM player_item WHERE item_id = 6;


-- Запрос 6. Пересечение. Найдите участников, которые совершали действия с X и совершали действия с Y, возможно в разные даты. Используйте A ∩ B и INTERSECT.
-- РА: π[player_id](σ[item_id=3](player_item)) ∩ π[player_id](σ[item_id=6](player_item))
-- Прогноз: Игроки, которые купили ОБА предмета одновременно: 1, 7, 9.
SELECT player_id FROM player_item WHERE item_id = 3
INTERSECT
SELECT player_id FROM player_item WHERE item_id = 6;


-- Запрос 7. Разность. Найдите участников, которые совершали действия с X, но не Y. Используйте A − B и EXCEPT.
-- РА: π[player_id](σ[item_id=3](player_item)) - π[player_id](σ[item_id=6](player_item))
-- Прогноз: Из множества А {1,7,9} вычитаем множество B {1,2,5,7,9,10}. Результат: пустое множество (0 строк).
SELECT player_id FROM player_item WHERE item_id = 3
EXCEPT
SELECT player_id FROM player_item WHERE item_id = 6;


-- Запрос 8. Отсутствие взаимодействий. Найдите участников, для которых нет ни одной записи в третьей таблице. Используйте разность всех идентификаторов из первой таблицы и идентификаторов из третьей таблицы; в SQL примените EXCEPT.
-- РА: π[player_id](player) - π[player_id](player_item)
-- Прогноз: Все игроки совершали покупки, однако игрок с player_id = 6 (PiccoloAuraFarmer) не совершал покупок Gold (item_id=1), но у него есть другие предметы. Проверим абсолютное отсутствие в player_item.
-- В наших INSERT-данных абсолютно все 10 игроков имеют хотя бы одну запись в player_item. Результат: 0 строк (пусто).
SELECT player_id FROM player
EXCEPT
SELECT player_id FROM player_item;

-- Исследование 1. Повторы. 
-- Выполнение SQL-запроса 2 без DISTINCT:
SELECT item_name FROM item;
-- Выполнение SQL-запроса 2 с DISTINCT:
SELECT DISTINCT item_name FROM item;
-- Объяснение: В таблице item все 11 названий предметов уникальны, поэтому оба запроса вернут одинаковое количество строк (11). 
-- Однако, если бы имена повторялись, вариант БЕЗ DISTINCT вернул бы мультимножество (bag), а вариант С DISTINCT строго соответствует операции проекции (π) реляционной алгебры, работающей исключительно с математическими множествами без дубликатов.


-- Исследование 2. Произведение и соединение.
-- Выберем 2 строки из первой таблицы (player_id IN (1, 2)) и 3 строки из третьей (player_item_id IN (1, 2, 5))
-- Прогноз числа строк Декартова произведения: 2 * 3 = 6 строк.

-- Часть А: Выражение алгебры (player × player_item) и выполнение CROSS JOIN:
SELECT p.player_id, p.username, pi.player_item_id, pi.player_id AS item_owner_id
FROM (SELECT * FROM player WHERE player_id IN (1, 2)) p
CROSS JOIN (SELECT * FROM player_item WHERE player_item_id IN (1, 2, 5)) pi;
-- Результат: Получено ровно 6 строк. Комбинируются все варианты, включая логически неверные.

-- Часть Б: Добавление условия равенства связанных идентификаторов:
SELECT p.player_id, p.username, pi.player_item_id, pi.player_id AS item_owner_id
FROM (SELECT * FROM player WHERE player_id IN (1, 2)) p
CROSS JOIN (SELECT * FROM player_item WHERE player_item_id IN (1, 2, 5)) pi
WHERE p.player_id = pi.player_id;
-- Объяснение изменения: Результат сократился с 6 строк до 3 строк. 
-- Условию фильтрации подошли только строки, где реальный владелец предмета совпал с игроком слева (Иванов связался со своими вещами id=1 и id=2, а GooGooGaGa связался со своей вещью id=5). Лишние декартовы комбинации отсеялись.


-- ============================================================================
-- ДОПОЛНИТЕЛЬНОЕ ЗАДАНИЕ (Логика условия «с каждым»)
-- ============================================================================
-- Множество объектов (предметов) второй таблицы: М = {'Sword' (id=3), 'Health Potion' (id=6)}
-- Логика "Хотя бы с одним" (Запрос 5 / UNION): Игрок должен купить или 3, или 6. Результат: 1, 2, 5, 7, 9, 10.
-- Логика "С каждым" (Деление отношений / INTERSECT): Игрок обязан купить И 3, И 6. Результат: 1, 7, 9.
--
-- Реализация схемы деления отношений через проекцию, произведение и разность на SQL (эквивалент деления P ÷ M):
SELECT player_id FROM player
EXCEPT
SELECT player_id FROM (
    -- Идеальные комбинации "все должны были купить всё из множества М"
    (SELECT p.player_id, i.item_id 
     FROM player p 
     CROSS JOIN (SELECT item_id FROM item WHERE item_id IN (3, 6)) i)
    EXCEPT
    -- Минус реальные факты покупок
    (SELECT player_id, item_id FROM player_item)
) as missing_purchases;
-- Результат: выведет ID игроков 1, 7, 9, у которых нет "пропущенных" покупок из целевого набора.
